cmake_minimum_required(VERSION 3.24)
project(RemoteControl LANGUAGES CXX)

string(TIMESTAMP REMOTE_CONTROL_BUILD_DATE "%Y/%m/%d")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(GTKSOURCEVIEW3 REQUIRED IMPORTED_TARGET gtksourceview-3.0)
find_package(OpenSSL REQUIRED)

set(GRCLIB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../grclib" CACHE PATH "Path to the GRClib source tree")
if(NOT EXISTS "${GRCLIB_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "GRClib source tree not found. Set GRCLIB_SOURCE_DIR to a checkout of https://github.com/MorenoLand/GScript.GRClib")
endif()
add_subdirectory("${GRCLIB_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/grclib")

add_library(remote_control_options src/RCOptions.cpp)
target_include_directories(remote_control_options PUBLIC src)

enable_testing()
add_executable(RCOptionsTests tests/RCOptionsTests.cpp)
target_link_libraries(RCOptionsTests PRIVATE remote_control_options)
add_test(NAME RCOptionsTests COMMAND RCOptionsTests)

add_executable(RemoteControl src/main.cpp src/Backup.cpp src/GScriptEditor.cpp src/RCAccounts.cpp src/TAccountsWindow.cpp src/TFileBrowser.cpp src/TFileBrowserTree.cpp src/TLocalBanWindow.cpp src/TNPCList.cpp src/TOptionsWindow.cpp src/TPlayerList.cpp src/TScriptList.cpp src/TRemoteFrame.cpp src/TServerList.cpp src/TServerTextEditor.cpp src/TToallsWindow.cpp src/TStartFrame.cpp)
if(WIN32)
    set_target_properties(RemoteControl PROPERTIES WIN32_EXECUTABLE TRUE)
    target_sources(RemoteControl PRIVATE assets/RemoteControl.rc)
endif()
target_link_libraries(RemoteControl PRIVATE remote_control_options grclib OpenSSL::Crypto OpenSSL::SSL PkgConfig::GTK3 PkgConfig::GTKSOURCEVIEW3)
target_compile_definitions(RemoteControl PRIVATE REMOTE_CONTROL_BUILD_DATE="${REMOTE_CONTROL_BUILD_DATE}")
if(WIN32)
    target_link_libraries(RemoteControl PRIVATE iphlpapi rpcrt4)
endif()
set_target_properties(RemoteControl PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")

foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(RemoteControl PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${configuration}" "${CMAKE_SOURCE_DIR}/bin")
endforeach()

if(WIN32)
    get_filename_component(MINGW_RUNTIME_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(MINGW_RUNTIME_PREFIX "${MINGW_RUNTIME_DIRECTORY}" DIRECTORY)
    set_tests_properties(RCOptionsTests PROPERTIES ENVIRONMENT "PATH=${MINGW_RUNTIME_DIRECTORY};$ENV{PATH}")
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:grclib>" "$<TARGET_FILE_DIR:RemoteControl>")
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} "-DREMOTE_CONTROL_EXECUTABLE=$<TARGET_FILE:RemoteControl>" "-DREMOTE_CONTROL_OUTPUT_DIRECTORY=$<TARGET_FILE_DIR:RemoteControl>" "-DREMOTE_CONTROL_MINGW_RUNTIME_DIRECTORY=${MINGW_RUNTIME_DIRECTORY}" -P "${CMAKE_SOURCE_DIR}/cmake/StageRuntimeDependencies.cmake")
    foreach(runtimeDataDirectory share/icons/Adwaita share/icons/hicolor share/gtk-3.0 share/glib-2.0/schemas)
        add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${MINGW_RUNTIME_PREFIX}/${runtimeDataDirectory}" "$<TARGET_FILE_DIR:RemoteControl>/${runtimeDataDirectory}")
    endforeach()
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RemoteControl>/certs")
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MINGW_RUNTIME_PREFIX}/etc/ssl/certs/ca-bundle.crt" "$<TARGET_FILE_DIR:RemoteControl>/certs/ca-bundle.crt")
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} "-DREMOTE_CONTROL_OUTPUT_DIRECTORY=$<TARGET_FILE_DIR:RemoteControl>" "-DREMOTE_CONTROL_MINGW_RUNTIME_DIRECTORY=${MINGW_RUNTIME_DIRECTORY}" -P "${CMAKE_SOURCE_DIR}/cmake/StageGtkPixbufLoaders.cmake")
elseif(APPLE)
    set_target_properties(RemoteControl PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:grclib>" "$<TARGET_FILE_DIR:RemoteControl>")
else()
    set_target_properties(RemoteControl PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
    add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:grclib>" "$<TARGET_FILE_DIR:RemoteControl>")
endif()

set(REMOTE_CONTROL_RUNTIME_SOURCE_DIR "${CMAKE_SOURCE_DIR}/assets/runtime" CACHE PATH "Path to Remote Control runtime assets")
add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} "-DREMOTE_CONTROL_RUNTIME_SOURCE_DIRECTORY=${REMOTE_CONTROL_RUNTIME_SOURCE_DIR}" "-DREMOTE_CONTROL_OUTPUT_DIRECTORY=$<TARGET_FILE_DIR:RemoteControl>" -P "${CMAKE_SOURCE_DIR}/cmake/StageRuntimeAssets.cmake")

add_custom_command(TARGET RemoteControl POST_BUILD COMMAND ${CMAKE_COMMAND} -E rm -f "$<TARGET_FILE_DIR:RemoteControl>/RemoteControl.ico")
